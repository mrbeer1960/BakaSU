open(kd+'/KpmBridge.kt','w').write(r'''package org.bakasu.bakasu.ui.screen.kpm

import android.content.Context
import com.topjohnwu.superuser.Shell
import java.io.File

/**
 * 内置 KPM 后端桥接（方案 B）：
 *  - kpatch 由 APK assets 提供，首次使用时释放到 /data/adb/ksu/bin/kpatch
 *    （app 私有目录被 SELinux 禁止 exec，这个目录可以）
 *  - 模块文件只放 /data/adb/kpm/kpm/，除此之外不留任何东西
 */
object KpmBridge {
    const val MODULE_DIR = "/data/adb/kpm/kpm"
    private const val BIN_DIR = "/data/adb/ksu/bin"
    private const val BIN = "$BIN_DIR/kpatch"

    @Volatile
    private var binReady = false

    fun ensureBin(context: Context): Boolean {
        if (binReady) return true
        val target = File(BIN)
        return try {
            if (!target.exists() || target.length() == 0L) {
                val dir = File(BIN_DIR)
                if (!dir.exists()) dir.mkdirs()
                context.assets.open("kpm/kpatch").use { input ->
                    target.outputStream().use { out -> input.copyTo(out) }
                }
                target.setExecutable(true, false)
            }
            binReady = target.exists()
            binReady
        } catch (t: Throwable) {
            target.exists()
        }
    }

    private fun run(cmd: String): String = try {
        val r = Shell.cmd(cmd).exec()
        (r.out + r.err).joinToString("\n").trim()
    } catch (t: Throwable) {
        t.message ?: "error"
    }

    fun hello(context: Context): Boolean {
        if (!ensureBin(context)) return false
        return run("$BIN hello") == "hello2026"
    }

    /** 已安装（管理器里放着的）模块名 */
    fun installed(): List<String> {
        val dir = File(MODULE_DIR)
        val files = dir.listFiles { f -> f.isFile && f.name.endsWith(".kpm") } ?: return emptyList()
        return files.map { it.name.removeSuffix(".kpm") }.sorted()
    }

    /** 内核里当前已加载的模块名 */
    fun loaded(): Set<String> = run("$BIN kpm list")
        .lines().map { it.trim() }.filter { it.isNotEmpty() }.toSet()

    fun load(name: String): String = run("$BIN kpm load '$MODULE_DIR/$name.kpm'")

    fun unload(name: String): String = run("$BIN kpm unload '$name'")

    fun info(name: String): String = run("$BIN kpm info '$name'")

    /** 控制模块（对应 kpatch kpm ctl0），参数为空时只打印帮助 */
    fun control(name: String, args: String): String =
        run("$BIN kpm ctl0 '$name' $args")

    /** 把用户选中的文件放进模块目录，返回模块名 */
    fun persist(context: Context, src: File): String? {
        val name = src.name.removeSuffix(".kpm").ifBlank { "module-${System.currentTimeMillis()}" }
        return try {
            val dir = File(MODULE_DIR)
            if (!dir.exists()) dir.mkdirs()
            src.copyTo(File(dir, "$name.kpm"), overwrite = true)
            name
        } catch (t: Throwable) {
            null
        }
    }

    fun remove(name: String): Boolean = try {
        File(MODULE_DIR, "$name.kpm").delete()
    } catch (t: Throwable) {
        false
    }
}
''')

/** 列表项：模块名 + 是否已加载 */
private data class KpmItem(val name: String, val loaded: Boolean)

@Composable
fun KpmPage(bottomPadding: Dp) {
    val context = LocalContext.current
    val scope = rememberCoroutineScope()

    var modules by remember { mutableStateOf(emptyList<KpmItem>()) }
    var busy by remember { mutableStateOf(false) }
    var output by remember { mutableStateOf("") }
    var showOutput by remember { mutableStateOf(false) }
    var controlTarget by remember { mutableStateOf<String?>(null) }
    var controlParam by remember { mutableStateOf("") }
    var refresh by remember { mutableStateOf(0) }

    LaunchedEffect(refresh) {
        val loadedNames = withContext(Dispatchers.IO) {
            if (KpmBridge.hello(context)) KpmBridge.loaded() else emptySet()
        }
        modules = withContext(Dispatchers.IO) {
            KpmBridge.installed().map { KpmItem(it, loadedNames.contains(it)) }
        }
    }

    val picker = rememberLauncherForActivityResult(ActivityResultContracts.OpenDocument()) { uri: Uri? ->
        if (uri != null) scope.launch {
            busy = true
            output = withContext(Dispatchers.IO) {
                try {
                    val tmp = File(context.cacheDir, "import-${System.currentTimeMillis()}.kpm")
                    context.contentResolver.openInputStream(uri)?.use { input ->
                        tmp.outputStream().use { out -> input.copyTo(out) }
                    }
                    val name = KpmBridge.persist(context, tmp) ?: return@withContext "保存失败"
                    "load: " + KpmBridge.load(name).ifBlank { "ok" }
                } catch (t: Throwable) {
                    "失败: ${t.message}"
                }
            }
            showOutput = true
            busy = false
            refresh++
        }
    }

    Box(Modifier.fillMaxSize().padding(bottom = bottomPadding)) {
        Column(Modifier.fillMaxSize()) {
            Row(
                Modifier.fillMaxWidth().padding(start = 20.dp, end = 8.dp, top = 8.dp),
                verticalAlignment = Alignment.CenterVertically,
            ) {
                Text("内核模块", style = MaterialTheme.typography.titleLarge, modifier = Modifier.weight(1f))
                IconButton(onClick = { refresh++ }) { Icon(Icons.Rounded.Refresh, "刷新") }
                IconButton(onClick = { picker.launch(arrayOf("*/*")) }) { Icon(Icons.Rounded.Add, "添加") }
            }

            if (busy) LinearProgressIndicator(Modifier.fillMaxWidth())

            if (modules.isEmpty()) {
                Box(Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                    Text(
                        "暂无内核模块",
                        style = MaterialTheme.typography.bodyMedium,
                        textAlign = TextAlign.Center,
                        color = MaterialTheme.colorScheme.onSurfaceVariant,
                    )
                }
            } else {
                LazyColumn(
                    Modifier.fillMaxSize(),
                    contentPadding = PaddingValues(start = 16.dp, end = 16.dp, top = 8.dp, bottom = 8.dp),
                    verticalArrangement = Arrangement.spacedBy(10.dp),
                ) {
                    items(modules) { module ->
                        Surface(
                            modifier = Modifier.fillMaxWidth(),
                            color = MaterialTheme.colorScheme.surface,
                            tonalElevation = 1.dp,
                            shape = RoundedCornerShape(20.dp),
                        ) {
                            Row(
                                Modifier.padding(16.dp),
                                verticalAlignment = Alignment.CenterVertically,
                            ) {

                                        else MaterialTheme.colorScheme.onSurfaceVariant,
