import android.content.Context
import com.topjohnwu.superuser.Shell
import java.io.File

/** 一个已安装的内核模块 */
data class KpmModule(
    val name: String,
    val version: String = "",
    val author: String = "",
    val args: String = "",
    val description: String = "",
    val loaded: Boolean = false,
    val embedded: Boolean = false,
)

/**
 * KPM 前端桥接：直接调 kpatch（和 APatch 走 JNI 直连内核是同一个思路）。
 * 注意：/data/adb 下的读写全部通过 root shell 完成，App 自身没有权限。
 */
object KpmBridge {
    private const val KPATCH = "/data/adb/ksu/bin/kpatch"
    const val MODULE_DIR = "/data/adb/kpm"
    private const val EMBED_WORK = "/data/local/tmp/kpm-chk"

    private fun sh(cmd: String): Pair<Int, String> = try {
        val r = Shell.cmd(cmd).exec()
        Pair(r.code, (r.out + r.err).joinToString("\n").trim())
    } catch (t: Throwable) {
        Pair(-1, t.message ?: "error")
    }

    private fun run(args: List<String>): String = sh((listOf(KPATCH) + args).joinToString(" ")).second

    fun available(): Boolean = run(listOf("hello")) == "hello2026"

    private var readyChecked = false
    private var readyCached = false
    private var readyBootId: String? = null
    private var appContext: Context? = null

    fun init(context: Context) {
        appContext = context.applicationContext
    }

    private fun currentBootId(): String? = try {
        File("/proc/sys/kernel/random/boot_id").readText().trim()
    } catch (t: Throwable) {
        null
    }

    private fun probe(): Boolean? = try {
        val r = Shell.cmd("$KPATCH hello").exec()
        val out = (r.out + r.err).joinToString("\n")
        out.contains("hello2026")
    } catch (t: Throwable) {
        null
    }

    private fun retryProbe(): Boolean? {
        var result: Boolean? = null
        var attempt = 0
        while (attempt < 3) {
            result = probe()
            if (result != null) return result
            attempt++
            if (attempt < 3) {
                try {
                    Thread.sleep(800)
                } catch (t: Throwable) {
                }
            }
        }
        return result
    }

    /** 开机进程调用：探测一次，把结果按 boot_id 写进管理器配置 */
    fun probeAndStore() {
        val bootId = currentBootId() ?: return
        val ctx = appContext ?: return
        val ok = retryProbe() ?: return
        ctx.getSharedPreferences("kpm_state", Context.MODE_PRIVATE)
            .edit()
            .putString("boot_id", bootId)
            .putBoolean("ready", ok)
            .commit()
    }

    /** 界面调用：只读上一次开机写好的结果，没有结果时不探测、不等待 */
    fun isKpmReady(): Boolean {
        val bootId = currentBootId() ?: return true
        val ctx = appContext ?: return true
        val sp = ctx.getSharedPreferences("kpm_state", Context.MODE_PRIVATE)
        val saved = sp.getString("boot_id", null)
        if (saved != null && saved == bootId) {
            return sp.getBoolean("ready", true)
        }
        return true
    }


    fun nums(): Int = run(listOf("kpm", "num")).trim().toIntOrNull() ?: 0

    private fun loadedNames(): Set<String> = run(listOf("kpm", "list"))
        .lines().map { it.trim() }.filter { it.isNotEmpty() }.toSet()


    private fun parseInfo(name: String, loaded: Set<String>): KpmModule {
        val fields = HashMap<String, String>()
        run(listOf("kpm", "info", name)).lines().forEach { line ->
            val idx = line.indexOf('=')
            if (idx > 0) fields[line.substring(0, idx).trim()] = line.substring(idx + 1).trim()
        }
        return KpmModule(
            name = name,
            version = fields["version"].orEmpty(),
            author = fields["author"].orEmpty(),
            args = fields["args"].orEmpty(),
            description = fields["description"].orEmpty(),
            loaded = loaded.contains(name),
        )
    }

    fun installed(context: Context): List<KpmModule> {
        val loaded = loadedNames()
        val embedded = embeddedNames(context)
        val names = (loaded + embedded).distinct().sorted()
        return names.map { n -> parseInfo(n, loaded).copy(embedded = embedded.contains(n)) }
    }

    /** 从任意路径加载（不落地，重启即失效） */
    fun load(path: String): String = run(listOf("kpm", "load", path))

    fun unload(name: String): String = run(listOf("kpm", "unload", name))

    fun info(name: String): String = run(listOf("kpm", "info", name))

    /** 用 kptools 读 .kpm 内部的模块名 */
    fun moduleName(path: String): String? =
        sh("/data/adb/ksu/bin/kptools -l -M '$path' 2>/dev/null").second
            .lines().firstOrNull { it.startsWith("name=") }
            ?.substringAfter("name=")?.trim()?.takeIf { it.isNotEmpty() }

    fun control(name: String, args: String): String =
        run(listOf("kpm", "ctl0", name, args)).ifBlank { "ok" }


    /** 直接读 boot 镜像，取里面真正嵌入的模块名（不再依赖本地清单） */
    fun embeddedNames(context: Context): Set<String> {
        ensureTools(context, listOf("kptools", "magiskboot"))
        val slot = sh("getprop ro.boot.slot_suffix").second.trim().ifBlank { "_a" }
        val out = sh(
            "W=$EMBED_WORK; rm -rf \$W && mkdir -p \$W && " +
                "dd if=/dev/block/by-name/boot$slot of=\$W/boot.img bs=4096 2>/dev/null && " +
                "cd \$W && /data/adb/ksu/bin/magiskboot unpack boot.img >/dev/null 2>&1 && " +
                "/data/adb/ksu/bin/kptools -l -i kernel 2>/dev/null; cd /; rm -rf \$W"
        ).second
        val names = out.lines()
            .map { it.trim() }
            .filter { it.startsWith("name=") }
            .map { it.removePrefix("name=").trim() }
            .filter { it.isNotEmpty() }
            .toSet()
        return names
    }


    /** 把 APK assets 里的二进制释放到 /data/adb/ksu/bin（只能走 root shell） */
    fun ensureTools(context: Context, assets: List<String>): String? {
        val cacheDir = File(context.cacheDir, "kpmbin")
        if (!cacheDir.exists()) cacheDir.mkdirs()
        for (name in assets) {
            val tmp = File(cacheDir, name)
            try {
                context.assets.open("kpm/$name").use { i ->
                    tmp.outputStream().use { o -> i.copyTo(o) }
                }
            } catch (t: Throwable) {
                return "读取 APK 内 $name 失败：${t.message}"
            }
            val (code, out) = sh(
                "mkdir -p /data/adb/ksu/bin && cp -f '${tmp.absolutePath}' '/data/adb/ksu/bin/$name' " +
                    "&& chmod 755 '/data/adb/ksu/bin/$name'"
            )
            if (code != 0) return "释放 $name 失败：$out"
        }
        return null
    }
}
''')

private fun displayName(context: Context, uri: Uri): String = try {
    context.contentResolver.query(uri, null, null, null, null)?.use { c ->
        val idx = c.getColumnIndex(OpenableColumns.DISPLAY_NAME)
        if (idx >= 0 && c.moveToFirst()) c.getString(idx) else ""

} catch (t: Throwable) {
    ""
}

private fun copyToCache(context: Context, uri: Uri): File? = try {
    val tmp = File(context.cacheDir, "import-${System.currentTimeMillis()}.kpm")
    context.contentResolver.openInputStream(uri)?.use { input ->
        tmp.outputStream().use { out -> input.copyTo(out) }
    }
    if (tmp.length() > 0) tmp else null

    null
}

import android.content.Context
import android.net.Uri
import com.topjohnwu.superuser.Shell
import java.io.File

/**
 * 真·嵌入：把 .kpm 写进 boot 镜像（APatch 的「嵌入」同思路）。
 *
 *   dd boot_<slot> → boot.img
 *   magiskboot unpack boot.img            → kernel
 *   kptools -p -i kernel -k kpimg -M <kpm> -T kpm -o kernel.new
 *   magiskboot repack boot.img            → new-boot.img
 *   dd new-boot.img → boot_<slot>
 *
 * 所有 /data/adb、/data/local/tmp 的读写都走 root shell（App 自身无权限）。
 * 不自动备份 boot（由用户自行负责）。
 */
object KpmEmbed {
    private const val BIN_DIR = "/data/adb/ksu/bin"
    private const val KTOOLS = "$BIN_DIR/kptools"
    private const val KPIMG = "$BIN_DIR/kpimg"
    private const val MAGISKBOOT = "$BIN_DIR/magiskboot"
    private const val WORK = "/tmp/kpm-embed"

    private fun sh(cmd: String): Pair<Int, String> = try {
        val r = Shell.cmd(cmd).exec()
        Pair(r.code, (r.out + r.err).joinToString("\n").trim())
    } catch (t: Throwable) {
        Pair(-1, t.message ?: "error")
    }

    private fun exists(path: String): Boolean = sh("[ -s '$path' ] && echo yes").second.contains("yes")

    fun embed(context: Context, moduleUri: Uri, onStep: (String) -> Unit = {}): String {
        val log = StringBuilder()
        fun step(msg: String) {
            log.append(msg).append('\n')
            onStep(msg)
        }

        KpmBridge.ensureTools(context, listOf("kptools", "kpimg", "magiskboot"))
            ?.let { return it }

        val slot = sh("getprop ro.boot.slot_suffix").second.trim().ifBlank { "_a" }
        val dev = "/dev/block/by-name/boot$slot"
        step("目标分区：$dev")

        val cacheKpm = File(context.cacheDir, "embed-${System.currentTimeMillis()}.kpm")
        try {
            context.contentResolver.openInputStream(moduleUri)?.use { i ->
                cacheKpm.outputStream().use { o -> i.copyTo(o) }
            } ?: return "读取所选文件失败"
        } catch (t: Throwable) {
            return "读取所选文件失败：${t.message}"
        }

        sh("rm -rf $WORK && mkdir -p $WORK")
        val kpm = "$WORK/module.kpm"
        var (code, out) = sh("cp -f '${cacheKpm.absolutePath}' '$kpm' && chmod 644 '$kpm'")
        step("准备模块文件：code=$code")
        if (code != 0) return log.toString() + "复制模块失败：$out"

        code = sh("dd if=$dev of=$WORK/boot.img bs=4096 2>&1").first
        step("读取 boot：code=$code")
        if (code != 0 || !exists("$WORK/boot.img")) return log.toString() + "读取 boot 失败"

        var r = sh("cd $WORK && $MAGISKBOOT unpack boot.img 2>&1")
        step("解包：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/kernel")) {
            return log.toString() + "解包失败：${r.second.take(400)}"
        }

        r = sh("cd $WORK && $KTOOLS -p -i kernel -k $KPIMG -M $kpm -T kpm -o kernel.new 2>&1")
        step("打补丁 + 嵌入：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/kernel.new")) {
            return log.toString() + "打补丁失败：${r.second.takeLast(600)}"
        }

        sh("cd $WORK && mv -f kernel.new kernel")
        sh("cd $WORK && rm -f new-boot.img")

        r = sh("cd $WORK && $MAGISKBOOT repack boot.img new-boot.img 2>&1")
        step("重打包：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/new-boot.img")) {
            return log.toString() + "重打包失败：${r.second.take(400)}"
        }

        r = sh("dd if=$WORK/new-boot.img of=$dev bs=4096 2>&1")
        step("写入分区：code=${r.first}")
        if (r.first != 0) return log.toString() + "写入失败：${r.second.take(400)}"

        sh("sync")
        sh("rm -rf $WORK")
        cacheKpm.delete()
        step("完成，重启后由内核直接加载该模块")
        return log.toString()
    }

/**

fun repatch(context: Context, moduleUris: List<Uri>, onStep: (String) -> Unit = {}): String {
    val log = StringBuilder()
    fun step(msg: String) {
        log.append(msg).append('\n')
        onStep(msg)
    }

    KpmBridge.ensureTools(context, listOf("kptools", "kpimg", "magiskboot"))?.let { return it }

    val slot = sh("getprop ro.boot.slot_suffix").second.trim().ifBlank { "_a" }
    val dev = "/dev/block/by-name/boot$slot"
    step("目标分区：$dev，保留模块 ${moduleUris.size} 个")

    sh("rm -rf $WORK && mkdir -p $WORK")

    val args = StringBuilder()
    val kept = mutableListOf<String>()
    moduleUris.forEachIndexed { index, uri ->
        val tmp = File(context.cacheDir, "keep-$index.kpm")
        try {
            context.contentResolver.openInputStream(uri)?.use { i ->
                tmp.outputStream().use { o -> i.copyTo(o) }
            }
        } catch (t: Throwable) {
            return "读取第 ${index + 1} 个文件失败：${t.message}"
        }
        val dst = "$WORK/keep-$index.kpm"
        val (c, _) = sh("cp -f '${tmp.absolutePath}' '$dst' && chmod 644 '$dst'")
        tmp.delete()
        if (c != 0) return log.toString() + "复制第 ${index + 1} 个模块失败"
        args.append("-M $dst -T kpm ")
        KpmBridge.moduleName(dst)?.let { kept.add(it) }
    }
    step("保留：" + kept.joinToString(", ").ifBlank { "（无，全部清空）" })

    var code = sh("dd if=$dev of=$WORK/boot.img bs=4096 2>&1").first
    step("读取 boot：code=$code")
    if (code != 0) return log.toString() + "读取 boot 失败"

    var r = sh("cd $WORK && $MAGISKBOOT unpack boot.img 2>&1")
    step("解包：code=${r.first}")
    if (r.first != 0) return log.toString() + "解包失败：${r.second.take(300)}"

    r = sh("cd $WORK && $KTOOLS -p -i kernel -k $KPIMG $args-o kernel.new 2>&1")
    step("重新打补丁：code=${r.first}")
    if (r.first != 0) return log.toString() + "打补丁失败：${r.second.takeLast(600)}"

    sh("cd $WORK && mv -f kernel.new kernel && rm -f new-boot.img")
    r = sh("cd $WORK && $MAGISKBOOT repack boot.img new-boot.img 2>&1")
    step("重打包：code=${r.first}")
    if (r.first != 0) return log.toString() + "重打包失败：${r.second.take(300)}"

    code = sh("dd if=$WORK/new-boot.img of=$dev bs=4096 2>&1").first
    step("写入分区：code=$code")
    if (code != 0) return log.toString() + "写入失败"

    sh("sync")
    sh("rm -rf $WORK")
    step("完成，重启后生效")
    return log.toString()
}

    /** 从 boot 里移除指定嵌入模块，其余保留（kptools -E 按名保留重建） */
    fun removeEmbedded(context: Context, name: String, onStep: (String) -> Unit = {}): String {
        val log = StringBuilder()
        fun step(msg: String) {
            log.append(msg).append('\n')
            onStep(msg)
        }

        KpmBridge.ensureTools(context, listOf("kptools", "kpimg", "magiskboot"))
            ?.let { return it }

        val slot = sh("getprop ro.boot.slot_suffix").second.trim().ifBlank { "_a" }
        val dev = "/dev/block/by-name/boot$slot"
        step("目标分区：$dev")

        sh("rm -rf $WORK && mkdir -p $WORK")
        var code = sh("dd if=$dev of=$WORK/boot.img bs=4096 2>&1").first
        step("读取 boot：code=$code")
        if (code != 0 || !exists("$WORK/boot.img")) return log.toString() + "读取 boot 失败"

        var r = sh("cd $WORK && $MAGISKBOOT unpack boot.img 2>&1")
        step("解包：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/kernel")) {
            return log.toString() + "解包失败：${r.second.take(400)}"
        }

        val extras = ArrayList<Pair<String, String>>()
        var inExtras = false
        var curType = "kpm"
        for (raw in sh("cd $WORK && $KTOOLS -l -i kernel 2>&1").second.lines()) {
            val line = raw.trim()
            if (line.startsWith("[extras]")) {
                inExtras = true
                continue
            }
            if (line.startsWith("[")) {
                if (!line.startsWith("[extra ")) inExtras = false
                continue
            }
            if (!inExtras) continue
            if (line.startsWith("type=")) curType = line.removePrefix("type=").trim()
            if (line.startsWith("name=")) {
                val n = line.removePrefix("name=").trim()
                if (n.isNotEmpty()) extras.add(n to curType)
            }
        }

        if (extras.isEmpty()) return log.toString() + "boot 里没有嵌入的模块"

        val keep = extras.filter { it.first != name }
        step("现有嵌入：" + extras.joinToString(", ") { it.first })
        step("保留：" + (if (keep.isEmpty()) "(none)" else keep.joinToString(", ") { it.first }))

        val keepArgs = keep.joinToString(" ") { "-E \"${it.first}\" -T ${it.second}" }
        val cmd = if (keep.isEmpty()) {
            "cd $WORK && $KTOOLS -p -i kernel -k $KPIMG -o kernel.new 2>&1"
        } else {
            "cd $WORK && $KTOOLS -p -i kernel -k $KPIMG $keepArgs -o kernel.new 2>&1"
        }
        r = sh(cmd)
        step("重建嵌入：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/kernel.new")) {
            return log.toString() + "移除失败：${r.second.takeLast(600)}"
        }

        sh("cd $WORK && mv -f kernel.new kernel && rm -f new-boot.img")
        r = sh("cd $WORK && $MAGISKBOOT repack boot.img new-boot.img 2>&1")
        step("重打包：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/new-boot.img")) {
            return log.toString() + "重打包失败：${r.second.take(400)}"
        }

        code = sh("dd if=$WORK/new-boot.img of=$dev bs=4096 2>&1").first
        step("写入分区：code=$code")
        if (code != 0) return log.toString() + "写入失败"

        sh("sync")
        sh("rm -rf $WORK")
        step("完成，重启后生效")
        return log.toString()
    }
}
