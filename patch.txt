import android.content.Context
import com.topjohnwu.superuser.Shell
import java.io.File

/** 一个已安装的内核模块 */
data class KpmModule(
    val name: String,
    val version: String = "",
    val author: String = "",
    val args: String = "",
    val description: String = "",
    val loaded: Boolean = false,
)

/**
 * KPM 前端桥接：直接调 kpatch（和 APatch 走 JNI 直连内核是同一个思路）。
 * 注意：/data/adb 下的读写全部通过 root shell 完成，App 自身没有权限。
 */
object KpmBridge {
    private const val KPATCH = "/data/adb/ksu/bin/kpatch"
    const val MODULE_DIR = "/data/adb/kpm"

    private fun sh(cmd: String): Pair<Int, String> = try {
        val r = Shell.cmd(cmd).exec()
        Pair(r.code, (r.out + r.err).joinToString("\n").trim())
    } catch (t: Throwable) {
        Pair(-1, t.message ?: "error")
    }

    private fun run(args: List<String>): String = sh((listOf(KPATCH) + args).joinToString(" ")).second

    fun available(): Boolean = run(listOf("hello")) == "hello2026"

    fun nums(): Int = run(listOf("kpm", "num")).trim().toIntOrNull() ?: 0

    private fun loadedNames(): Set<String> = run(listOf("kpm", "list"))
        .lines().map { it.trim() }.filter { it.isNotEmpty() }.toSet()

    /** 模块名列表（ls 走 root shell） */
    fun installedNames(): List<String> = sh("ls -1 $MODULE_DIR 2>/dev/null").second
        .lines().map { it.trim() }
        .filter { it.endsWith(".kpm") }
        .map { it.removeSuffix(".kpm") }
        .sorted()

    private fun parseInfo(name: String, loaded: Set<String>): KpmModule {
        val fields = HashMap<String, String>()
        run(listOf("kpm", "info", name)).lines().forEach { line ->
            val idx = line.indexOf('=')
            if (idx > 0) fields[line.substring(0, idx).trim()] = line.substring(idx + 1).trim()
        }
        return KpmModule(
            name = name,
            version = fields["version"].orEmpty(),
            author = fields["author"].orEmpty(),
            args = fields["args"].orEmpty(),
            description = fields["description"].orEmpty(),
            loaded = loaded.contains(name),
        )
    }

    fun installed(): List<KpmModule> {
        val loaded = loadedNames()
        return installedNames().map { parseInfo(it, loaded) }
    }

    fun load(name: String): String = run(listOf("kpm", "load", "$MODULE_DIR/$name.kpm"))

    fun unload(name: String): String = run(listOf("kpm", "unload", name))

    fun info(name: String): String = run(listOf("kpm", "info", name))

    fun control(name: String, args: String): String =
        run(listOf("kpm", "ctl0", name, args)).ifBlank { "ok" }

    /** 把用户选的临时文件复制进模块目录（用 root shell 拷） */
    fun persist(src: File): String? {
        val name = src.name.removeSuffix(".kpm").ifBlank { "module-${System.currentTimeMillis()}" }
        val dst = "$MODULE_DIR/$name.kpm"
        val (code, out) = sh("mkdir -p $MODULE_DIR && cp -f '${src.absolutePath}' '$dst' && chmod 644 '$dst'")
        return if (code == 0) name else null.also { println(out) }
    }

    fun remove(name: String): Boolean =
        sh("rm -f '$MODULE_DIR/$name.kpm'").first == 0

    /** 把 APK assets 里的二进制释放到 /data/adb/ksu/bin（只能走 root shell） */
    fun ensureTools(context: Context, assets: List<String>): String? {
        val cacheDir = File(context.cacheDir, "kpmbin")
        if (!cacheDir.exists()) cacheDir.mkdirs()
        for (name in assets) {
            val tmp = File(cacheDir, name)
            try {
                context.assets.open("kpm/$name").use { i ->
                    tmp.outputStream().use { o -> i.copyTo(o) }
                }
            } catch (t: Throwable) {
                return "读取 APK 内 $name 失败：${t.message}"
            }
            val (code, out) = sh(
                "mkdir -p /data/adb/ksu/bin && cp -f '${tmp.absolutePath}' '/data/adb/ksu/bin/$name' " +
                    "&& chmod 755 '/data/adb/ksu/bin/$name'"
            )
            if (code != 0) return "释放 $name 失败：$out"
        }
        return null
    }
}
''')

import android.content.Context
import android.net.Uri
import com.topjohnwu.superuser.Shell
import java.io.File

/**
 * 真·嵌入：把 .kpm 写进 boot 镜像（APatch 的「嵌入」同思路）。
 *
 *   dd boot_<slot> → boot.img
 *   magiskboot unpack boot.img            → kernel
 *   kptools -p -i kernel -k kpimg -M <kpm> -T kpm -o kernel.new
 *   magiskboot repack boot.img            → new-boot.img
 *   dd new-boot.img → boot_<slot>
 *
 * 所有 /data/adb、/data/local/tmp 的读写都走 root shell（App 自身无权限）。
 * 不自动备份 boot（由用户自行负责）。
 */
object KpmEmbed {
    private const val BIN_DIR = "/data/adb/ksu/bin"
    private const val KTOOLS = "$BIN_DIR/kptools"
    private const val KPIMG = "$BIN_DIR/kpimg"
    private const val MAGISKBOOT = "$BIN_DIR/magiskboot"
    private const val WORK = "/data/local/tmp/kpm-embed"

    private fun sh(cmd: String): Pair<Int, String> = try {
        val r = Shell.cmd(cmd).exec()
        Pair(r.code, (r.out + r.err).joinToString("\n").trim())
    } catch (t: Throwable) {
        Pair(-1, t.message ?: "error")
    }

    private fun exists(path: String): Boolean = sh("[ -s '$path' ] && echo yes").second.contains("yes")

    fun embed(context: Context, moduleUri: Uri, onStep: (String) -> Unit = {}): String {
        val log = StringBuilder()
        fun step(msg: String) {
            log.append(msg).append('\n')
            onStep(msg)
        }

        KpmBridge.ensureTools(context, listOf("kptools", "kpimg", "magiskboot"))
            ?.let { return it }

        val slot = sh("getprop ro.boot.slot_suffix").second.trim().ifBlank { "_a" }
        val dev = "/dev/block/by-name/boot$slot"
        step("目标分区：$dev")

        val cacheKpm = File(context.cacheDir, "embed-${System.currentTimeMillis()}.kpm")
        try {
            context.contentResolver.openInputStream(moduleUri)?.use { i ->
                cacheKpm.outputStream().use { o -> i.copyTo(o) }
            } ?: return "读取所选文件失败"
        } catch (t: Throwable) {
            return "读取所选文件失败：${t.message}"
        }

        sh("rm -rf $WORK && mkdir -p $WORK")
        val kpm = "$WORK/module.kpm"
        var (code, out) = sh("cp -f '${cacheKpm.absolutePath}' '$kpm' && chmod 644 '$kpm'")
        step("准备模块文件：code=$code")
        if (code != 0) return log.toString() + "复制模块失败：$out"

        code = sh("dd if=$dev of=$WORK/boot.img bs=4096 2>&1").first
        step("读取 boot：code=$code")
        if (code != 0 || !exists("$WORK/boot.img")) return log.toString() + "读取 boot 失败"

        var r = sh("cd $WORK && $MAGISKBOOT unpack boot.img 2>&1")
        step("解包：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/kernel")) {
            return log.toString() + "解包失败：${r.second.take(400)}"
        }

        r = sh("cd $WORK && $KTOOLS -p -i kernel -k $KPIMG -M $kpm -T kpm -o kernel.new 2>&1")
        step("打补丁 + 嵌入：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/kernel.new")) {
            return log.toString() + "打补丁失败：${r.second.takeLast(600)}"
        }

        sh("cd $WORK && mv -f kernel.new kernel")
        sh("cd $WORK && rm -f new-boot.img")

        r = sh("cd $WORK && $MAGISKBOOT repack boot.img new-boot.img 2>&1")
        step("重打包：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/new-boot.img")) {
            return log.toString() + "重打包失败：${r.second.take(400)}"
        }

        r = sh("dd if=$WORK/new-boot.img of=$dev bs=4096 2>&1")
        step("写入分区：code=${r.first}")
        if (r.first != 0) return log.toString() + "写入失败：${r.second.take(400)}"

        sh("sync")
        cacheKpm.delete()
        step("完成，重启后由内核直接加载该模块")
        return log.toString()
    }
}
''')
