import android.content.Context
import com.topjohnwu.superuser.Shell
import java.io.File

/** 一个已安装的内核模块 */
data class KpmModule(
    val name: String,
    val version: String = "",
    val author: String = "",
    val args: String = "",
    val description: String = "",
    val loaded: Boolean = false,
    val embedded: Boolean = false,
)

/**
 * KPM 前端桥接。
 *
 * 加载 / 卸载 / 列表 / 信息 / 控制：统一走 root shell（ksud kpm → kpatch）。
 * 不走 JNI：App 普通 uid 直接发 supercall 会被内核拒（实测 num=-1、list 空、hello 空）。
 * 只读内核分区、释放工具：同样通过 root shell。
 */
object KpmBridge {
    const val MODULE_DIR = "/data/adb/kpm"
    private const val KSUD = "/data/adb/ksud"
    private const val EMBED_WORK = "/data/local/tmp/kpm-chk"

    private fun sh(cmd: String): Pair<Int, String> = try {
        val r = Shell.cmd(cmd).exec()
        Pair(r.code, (r.out + r.err).joinToString("\n").trim())
    } catch (t: Throwable) {
        Pair(-1, t.message ?: "error")
    }

    /** ksud kpm <args> → kpatch kpm <args>（root shell） */
    private fun kpm(vararg args: String): Pair<Int, String> =
        sh((listOf(KSUD, "kpm") + args).joinToString(" "))

    /** JNI 的 list/info 会打印到 stdout，这里重定向到临时文件再读回 */
    private fun capture(context: Context, block: (String) -> Int): Pair<Int, String> {
        val f = File(context.cacheDir, "kpm-out-${System.nanoTime()}.txt")
        val rc = try {
            block(f.absolutePath)
        } catch (t: Throwable) {
            -1
        }
        val text = try {
            f.readText()
        } catch (t: Throwable) {
            ""
        }
        f.delete()
        return rc to text
    }

    fun nums(): Int = kpm("num").second.trim().toIntOrNull() ?: 0

    private fun loadedNames(context: Context): Set<String> =
        kpm("list").second.lines().map { it.trim() }.filter { it.isNotEmpty() }.toSet()

    private fun parseInfo(context: Context, name: String, loaded: Set<String>): KpmModule {
        val fields = HashMap<String, String>()
        val text = kpm("info", name).second
        text.lines().forEach { line ->
            val idx = line.indexOf('=')
            if (idx > 0) {
                fields[line.substring(0, idx).trim()] = line.substring(idx + 1).trim()
            }
        }
        return KpmModule(
            name = name,
            version = fields["version"].orEmpty(),
            author = fields["author"].orEmpty(),
            args = fields["args"].orEmpty(),
            description = fields["description"].orEmpty(),
            loaded = loaded.contains(name),
        )
    }

    fun installed(context: Context): List<KpmModule> {
        val loaded = loadedNames(context)
        val embedded = embeddedNames(context)
        val names = (loaded + embedded).distinct().sorted()
        return names.map { n -> parseInfo(context, n, loaded).copy(embedded = embedded.contains(n)) }
    }

    /** 从任意路径加载（不落地，重启即失效） */
    fun load(path: String): String {
        val (rc, text) = kpm("load", "'$path'")
        return if (rc == 0) text else "失败(code=$rc)：$text"
    }

    fun unload(name: String): String {
        val (rc, text) = kpm("unload", name)
        return if (rc == 0) text else "失败(code=$rc)：$text"
    }

    fun info(context: Context, name: String): String = kpm("info", name).second

    /** 用 kptools 读 .kpm 内部的模块名 */
    fun moduleName(path: String): String? =
        sh("/data/adb/ksu/bin/kptools -l -M '$path' 2>/dev/null").second
            .lines().firstOrNull { it.startsWith("name=") }
            ?.substringAfter("name=")?.trim()?.takeIf { it.isNotEmpty() }

    fun control(name: String, args: String): String {
        val (rc, text) = kpm("ctl0", name, args)
        return if (rc == 0) text.ifBlank { "ok" } else text.ifBlank { "error: $rc" }
    }

    /** 直接读 boot 镜像，取里面真正嵌入的模块名（不依赖本地清单） */
    fun embeddedNames(context: Context): Set<String> {
        ensureTools(context, listOf("kptools", "magiskboot"))
        val slot = sh("getprop ro.boot.slot_suffix").second.trim().ifBlank { "_a" }
        val out = sh(
            "W=$EMBED_WORK; rm -rf \$W && mkdir -p \$W && " +
                "dd if=/dev/block/by-name/boot$slot of=\$W/boot.img bs=4096 2>/dev/null && " +
                "cd \$W && /data/adb/ksu/bin/magiskboot unpack boot.img >/dev/null 2>&1 && " +
                "/data/adb/ksu/bin/kptools -l -i kernel 2>/dev/null; cd /; rm -rf \$W"
        ).second
        return out.lines()
            .map { it.trim() }
            .filter { it.startsWith("name=") }
            .map { it.removePrefix("name=").trim() }
            .filter { it.isNotEmpty() }
            .toSet()
    }

    /** 把 APK assets 里的二进制释放到 /data/adb/ksu/bin（只能走 root shell） */
    fun ensureTools(context: Context, assets: List<String>): String? {
        val cacheDir = File(context.cacheDir, "kpmbin")
        if (!cacheDir.exists()) cacheDir.mkdirs()
        for (name in assets) {
            val tmp = File(cacheDir, name)
            try {
                context.assets.open("kpm/$name").use { i ->
                    tmp.outputStream().use { o -> i.copyTo(o) }
                }
            } catch (t: Throwable) {
                return "读取 APK 内 $name 失败：${t.message}"
            }
            val (code, out) = sh(
                "mkdir -p /data/adb/ksu/bin && cp -f '${tmp.absolutePath}' '/data/adb/ksu/bin/$name' " +
                    "&& chmod 755 '/data/adb/ksu/bin/$name'"
            )
            if (code != 0) return "释放 $name 失败：$out"
        }
        return null
    }
}

private fun displayName(context: Context, uri: Uri): String = try {
    context.contentResolver.query(uri, null, null, null, null)?.use { c ->
        val idx = c.getColumnIndex(OpenableColumns.DISPLAY_NAME)
        if (idx >= 0 && c.moveToFirst()) c.getString(idx) else ""
    } ?: ""
} catch (t: Throwable) {
    ""
}

private fun copyToCache(context: Context, uri: Uri): File? = try {
    val tmp = File(context.cacheDir, "import-${System.currentTimeMillis()}.kpm")
    context.contentResolver.openInputStream(uri)?.use { input ->
        tmp.outputStream().use { out -> input.copyTo(out) }
    }
    if (tmp.length() > 0) tmp else null
} catch (t: Throwable) {
    null
}

/**
 * 真·嵌入：把 .kpm 写进 boot 镜像（APatch 的「嵌入」同思路）。
 *
 *   dd boot_<slot> → boot.img
 *   magiskboot unpack boot.img            → kernel
 *   kptools -p -i kernel -k kpimg -M <kpm> -T kpm -o kernel.new
 *   magiskboot repack boot.img            → new-boot.img
 *   dd new-boot.img → boot_<slot>
 *

 * 不自动备份 boot（由用户自行负责）。

    private const val KTOOLS = "$BIN_DIR/kptools"

    private fun sh(cmd: String): Pair<Int, String> = try {
        val r = Shell.cmd(cmd).exec()
        Pair(r.code, (r.out + r.err).joinToString("\n").trim())
    } catch (t: Throwable) {
        Pair(-1, t.message ?: "error")

    private fun exists(path: String): Boolean = sh("[ -s '$path' ] && echo yes").second.contains("yes")

        fun step(msg: String) {
            log.append(msg).append('\n')
            onStep(msg)
        }

    /** 从 boot 里移除指定嵌入模块，其余保留（kptools -E 按名保留重建） */
    fun removeEmbedded(context: Context, name: String, onStep: (String) -> Unit = {}): String {
        val log = StringBuilder()
        fun step(msg: String) {
            log.append(msg).append('\n')
            onStep(msg)
        }

        KpmBridge.ensureTools(context, listOf("kptools", "kpimg", "magiskboot"))
            ?.let { return it }

        val slot = sh("getprop ro.boot.slot_suffix").second.trim().ifBlank { "_a" }
        val dev = "/dev/block/by-name/boot$slot"
        step("目标分区：$dev")

        sh("rm -rf $WORK && mkdir -p $WORK")
        var code = sh("dd if=$dev of=$WORK/boot.img bs=4096 2>&1").first
        step("读取 boot：code=$code")
        if (code != 0 || !exists("$WORK/boot.img")) return log.toString() + "读取 boot 失败"

        var r = sh("cd $WORK && $MAGISKBOOT unpack boot.img 2>&1")
        step("解包：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/kernel")) {
            return log.toString() + "解包失败：${r.second.take(400)}"
        }

        val extras = ArrayList<Pair<String, String>>()
        var inExtras = false
        var curType = "kpm"
        for (raw in sh("cd $WORK && $KTOOLS -l -i kernel 2>&1").second.lines()) {
            val line = raw.trim()
            if (line.startsWith("[extras]")) {
                inExtras = true
                continue
            }
            if (line.startsWith("[")) {
                if (!line.startsWith("[extra ")) inExtras = false
                continue
            }
            if (!inExtras) continue
            if (line.startsWith("type=")) curType = line.removePrefix("type=").trim()
            if (line.startsWith("name=")) {
                val n = line.removePrefix("name=").trim()
                if (n.isNotEmpty()) extras.add(n to curType)
            }
        }

        if (extras.isEmpty()) return log.toString() + "boot 里没有嵌入的模块"

        val keep = extras.filter { it.first != name }
        step("现有嵌入：" + extras.joinToString(", ") { it.first })
        step("保留：" + (if (keep.isEmpty()) "(none)" else keep.joinToString(", ") { it.first }))

        val keepArgs = keep.joinToString(" ") { "-E \"${it.first}\" -T ${it.second}" }
        val cmd = if (keep.isEmpty()) {
            "cd $WORK && $KTOOLS -p -i kernel -k $KPIMG -o kernel.new 2>&1"
        } else {
            "cd $WORK && $KTOOLS -p -i kernel -k $KPIMG $keepArgs -o kernel.new 2>&1"
        }
        r = sh(cmd)
        step("重建嵌入：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/kernel.new")) {
            return log.toString() + "移除失败：${r.second.takeLast(600)}"
        }

        sh("cd $WORK && mv -f kernel.new kernel && rm -f new-boot.img")
        r = sh("cd $WORK && $MAGISKBOOT repack boot.img new-boot.img 2>&1")
        step("重打包：code=${r.first}")
        if (r.first != 0 || !exists("$WORK/new-boot.img")) {
            return log.toString() + "重打包失败：${r.second.take(400)}"

        code = sh("dd if=$WORK/new-boot.img of=$dev bs=4096 2>&1").first

        sh("sync")
        sh("rm -rf $WORK")
        step("完成，重启后生效")
        return log.toString()
    }
