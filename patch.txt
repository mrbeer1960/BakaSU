if i>0:
    g=s[i:i+700].replace('disabledContainerColor','containerColor').replace('disabledContentColor','contentColor')
    s=s[:i]+g+s[i+700:]
L=s.split('\n');o=[];k=0

    if 'showReplacingSnackbar(' in L[k] and 'val result' in L[k]:
        d=L[k].count('(')-L[k].count(')');k+=1
        while k<len(L) and d>0:
            d+=L[k].count('(')-L[k].count(')');k+=1
        while k<len(L) and L[k].strip()=='':k+=1
        if k<len(L) and 'SnackbarResult.ActionPerformed' in L[k]:
            d=L[k].count('{')-L[k].count('}');k+=1
            while k<len(L) and d>0:
                d+=L[k].count('{')-L[k].count('}');k+=1
        continue
    o.append(L[k]);k+=1
L=o;o=[];k=0
while k<len(L):
    if 'val formatter = if (module.metamodule)' in L[k]:
        j=k
        while 'ConfirmResult.Confirmed' not in L[j]:j+=1
        d=L[j].count('{')-L[j].count('}');j+=1
        while d>0:
            d+=L[j].count('{')-L[j].count('}');j+=1
        k=j;continue
    o.append(L[k]);k+=1
