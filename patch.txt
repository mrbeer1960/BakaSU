import com.topjohnwu.superuser.Shell
import java.io.File

/** 一个已安装的内核模块 */
data class KpmModule(
    val name: String,
    val version: String = "",
    val author: String = "",
    val args: String = "",
    val description: String = "",
    val loaded: Boolean = false,
)

/**
 * KPM 前端桥接：全部走 ksud 的 kpm 子命令（ksud 直接和内核里的 kpatch 通信）。
 * 不再需要任何外部二进制。
 */
object KpmBridge {
    private const val KSUD = "/data/adb/ksud"
    const val MODULE_DIR = "/data/adb/kpm"

    private fun run(args: List<String>): String = try {
        val r = Shell.cmd((listOf(KSUD) + args).joinToString(" ")).exec()
        (r.out + r.err).joinToString("\n").trim()
    } catch (t: Throwable) {
        t.message ?: "error"
    }

    /** 内核里是否有可用的 kpatch（ksud kpm nums 能跑通即算） */
    fun available(): Boolean = try {
        val out = run(listOf("kpm", "nums"))
        out.isNotEmpty() && !out.contains("failed") && !out.contains("not")
    } catch (t: Throwable) {
        false
    }

    private fun loadedNames(): Set<String> = run(listOf("kpm", "list"))
        .lines().map { it.trim() }.filter { it.isNotEmpty() }.toSet()

    /** 解析 ksud kpm info 的 key=value 输出 */
    private fun parseInfo(name: String, loaded: Set<String>): KpmModule {
        val fields = HashMap<String, String>()
        run(listOf("kpm", "info", name)).lines().forEach { line ->
            val idx = line.indexOf('=')
            if (idx > 0) {
                fields[line.substring(0, idx).trim()] = line.substring(idx + 1).trim()
            }
        }
        return KpmModule(
            name = name,
            version = fields["version"].orEmpty(),
            author = fields["author"].orEmpty(),
            args = fields["args"].orEmpty(),
            description = fields["description"].orEmpty(),
            loaded = loaded.contains(name),
        )
    }

    /** 已安装（/data/adb/kpm 里放着）的模块列表 */
    fun installed(): List<KpmModule> {
        val dir = File(MODULE_DIR)
        val files = dir.listFiles { f -> f.isFile && f.name.endsWith(".kpm") } ?: return emptyList()
        val loaded = loadedNames()
        return files.map { parseInfo(it.name.removeSuffix(".kpm"), loaded) }.sortedBy { it.name }
    }

    fun load(name: String): String = run(listOf("kpm", "load", "$MODULE_DIR/$name.kpm"))

    fun unload(name: String): String = run(listOf("kpm", "unload", name))

    fun control(name: String, args: String): String =
        run(listOf("kpm", "control", name, args)).ifBlank { "ok" }

    /** 把用户选的文件放进模块目录，返回模块名 */
    fun persist(src: File): String? {
        val name = src.name.removeSuffix(".kpm").ifBlank { "module-${System.currentTimeMillis()}" }
        return try {
            val dir = File(MODULE_DIR)
            if (!dir.exists()) dir.mkdirs()
            src.copyTo(File(dir, "$name.kpm"), overwrite = true)
            name
        } catch (t: Throwable) {
            null
        }
    }

    fun remove(name: String): Boolean = try {
        File(MODULE_DIR, "$name.kpm").delete()
    } catch (t: Throwable) {
        false
    }
}
''')

//!
//! No helper binary is involved: everything goes through KernelPatch's
//! "supercall" (syscall 45). Root/su uid is enough, no superkey required.

use anyhow::{anyhow, Result};
use libc::{c_char, c_long, syscall};
use log::{info, warn};
use std::ffi::CString;
use std::fs;

/// KernelPatch supercall syscall number
const NR_SUPERCALL: c_long = 45;

const CMD_LOAD: c_long = 0x1020;
const CMD_UNLOAD: c_long = 0x1021;
const CMD_CONTROL: c_long = 0x1022;
const CMD_NUMS: c_long = 0x1030;
const CMD_LIST: c_long = 0x1031;
const CMD_INFO: c_long = 0x1032;

/// Must match the kpatch build inside the kernel (KPatch-Next version)
const KP_MAJOR: u32 = 0;
const KP_MINOR: u32 = 13;
const KP_PATCH: u32 = 5;

/// directory that holds *.kpm modules
const KPM_DIR: &str = "/data/adb/kpm";

fn ver_and_cmd(cmd: c_long) -> c_long {
    let v = ((KP_MAJOR << 16) | (KP_MINOR << 8) | KP_PATCH) as c_long;
    (v << 32) | (0x1158 << 16) | (cmd & 0xFFFF)
}

fn sc(cmd: c_long, a1: c_long, a2: c_long, a3: c_long) -> c_long {
    let key = CString::new("su").unwrap_or_default();
    unsafe { syscall(NR_SUPERCALL, key.as_ptr(), ver_and_cmd(cmd), a1, a2, a3) as c_long }
}

fn loaded_list() -> Vec<String> {
    let mut buf = vec![0u8; 8192];
    let ret = sc(
        CMD_LIST,
        buf.as_mut_ptr() as c_long,
        buf.len() as c_long,
        0,
    );
    if ret < 0 {
        return Vec::new();
    }
    let len = buf.iter().position(|&b| b == 0).unwrap_or(0);
    String::from_utf8_lossy(&buf[..len])
        .lines()
        .map(|l| l.trim().to_string())
        .filter(|l| !l.is_empty())
        .collect()
}

pub fn nums() -> Result<()> {
    let n = sc(CMD_NUMS, 0, 0, 0);
    if n < 0 {
        return Err(anyhow!("KPM_NUMS failed: {n}"));
    }
    println!("{n}");
    Ok(())
}

pub fn list() -> Result<()> {
    let list = loaded_list();
    for name in list {
        println!("{name}");
    }
    Ok(())
}

pub fn load(path: &str) -> Result<()> {
    let c_path = CString::new(path).map_err(|_| anyhow!("invalid path"))?;
    let ret = sc(CMD_LOAD, c_path.as_ptr() as c_long, 0, 0);
    if ret < 0 {
        return Err(anyhow!("KPM_LOAD failed: {ret}"));
    }
    Ok(())
}

pub fn unload(name: &str) -> Result<()> {
    let c_name = CString::new(name).map_err(|_| anyhow!("invalid name"))?;
    let ret = sc(CMD_UNLOAD, c_name.as_ptr() as c_long, 0, 0);
    if ret < 0 {
        return Err(anyhow!("KPM_UNLOAD failed: {ret}"));
    }
    Ok(())
}

pub fn info(name: &str) -> Result<()> {
    let c_name = CString::new(name).map_err(|_| anyhow!("invalid name"))?;
    let mut buf = vec![0u8; 4096];
    let ret = sc(
        CMD_INFO,
        c_name.as_ptr() as c_long,
        buf.as_mut_ptr() as c_long,
        buf.len() as c_long,
    );
    if ret < 0 {
        return Err(anyhow!("KPM_INFO failed: {ret}"));
    }
    let len = buf.iter().position(|&b| b == 0).unwrap_or(0);
    print!("{}", String::from_utf8_lossy(&buf[..len]));
    Ok(())
}

pub fn control(name: &str, args: &str) -> Result<()> {
    let c_name = CString::new(name).map_err(|_| anyhow!("invalid name"))?;
    let c_args = CString::new(args).unwrap_or_default();
    let mut out = vec![0u8; 4096];
    let ret = sc(
        CMD_CONTROL,
        c_name.as_ptr() as c_long,
        c_args.as_ptr() as c_long,
        out.as_mut_ptr() as c_long,
    );
    let len = out.iter().position(|&b| b == 0).unwrap_or(0);
    let text = String::from_utf8_lossy(&out[..len]).to_string();
    if !text.is_empty() {
        print!("{text}");
    }
    if ret < 0 {
        return Err(anyhow!("KPM_CONTROL failed: {ret}"));
    }
    Ok(())
}

/// Called from the `services` boot stage so ksud owns kpm autoload.

    let loaded = loaded_list();

        Ok(e) => e,
        Err(_) => return Ok(()),
    };

        let path = entry.path();
        if path.extension().and_then(|e| e.to_str()) != Some("kpm") {
            continue;
        }
        let name = path
            .file_stem()
            .and_then(|s| s.to_str())
            .unwrap_or("")
            .to_string();
        if name.is_empty() || loaded.iter().any(|l| *l == name) {
            continue;
        }
        match load(path.to_str().unwrap_or("")) {
            Ok(()) => info!("kpm: loaded {name}"),
            Err(e) => warn!("kpm: load {name} failed: {e}"),
        }
    }
    Ok(())
